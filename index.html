// চ্যানেল ক্লিক করলে প্লেয়ারে ভিডিও লোড হওয়ার ফাংশন document.querySelectorAll('.channel-data').forEach(item => { item.addEventListener('click', function() { let url = this.getAttribute('data-url'); let playerBox = document.getElementById('player-container'); // প্রক্সি লিংকের মাধ্যমে স্ট্রিম রাউট করা let proxyStreamUrl = `/api/proxy?url=${encodeURIComponent(url)}`; playerBox.innerHTML = ` `; const video = document.getElementById('video-player'); // ব্রাউজারে নেটিভ HLS সাপোর্ট (যেমন Safari) থাকলে সরাসরি চালাবে, // না হলে সাধারণ সোর্স হিসেবে সেট করবে if (video.canPlayType('application/vnd.apple.mpegurl')) { video.src = proxyStreamUrl; } else { // সাধারণ ভিডিও ট্যাগ অথবা Hls.js ইন্টিগ্রেশন video.src = proxyStreamUrl; } }); });
